                values[i+1]=x[base+i+1]/16.0f;

                values[i+3]=x[base+i+3]/4096.0f;

                const device ushort* words=(const device ushort*)(w+row*(K/8)+base/8);

                          +values[4*i+2]*(words[i]&0x0f00)

                }
                const float s=float(scales[row*(K/64)+base/64]);

            }
